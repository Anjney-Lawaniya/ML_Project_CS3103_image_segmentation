===== src/audit.py =====
"""Audit training images and masks and summarize road/background pixel balance."""

from __future__ import annotations

import argparse
import logging
from pathlib import Path

import numpy as np
from PIL import Image, UnidentifiedImageError
from tqdm import tqdm

LOGGER = logging.getLogger(__name__)
IMAGE_DIRECTORY = Path("tiff") / "train"
MASK_DIRECTORY = Path("tiff") / "train_labels"
VALID_IMAGE_SUFFIXES = {".tif", ".tiff", ".png", ".jpg", ".jpeg"}


def _resolve_pair(raw_dir: Path, filename: str) -> tuple[Path, Path]:
    """Resolve a manifest image filename to its expected image and mask paths."""
    image_path = raw_dir / IMAGE_DIRECTORY / filename
    mask_dir = raw_dir / MASK_DIRECTORY
    mask_candidates = (
        (
            candidate
            for candidate in mask_dir.iterdir()
            if candidate.is_file()
            and candidate.suffix.casefold() in VALID_IMAGE_SUFFIXES
            and candidate.stem.casefold() == image_path.stem.casefold()
        )
        if mask_dir.is_dir()
        else iter(())
    )
    mask_path = next(mask_candidates, mask_dir / f"{image_path.stem}.tif")
    return image_path, mask_path


def audit_training_data(
    data_dir: str = "data/raw",
    splits_dir: str = "data/splits",
    report_path: str = "results/tables/data_audit_summary.txt",
) -> dict[str, int | float]:
    """Audit every listed training pair and save pixel counts and percentages.

    Only ``train.txt`` is read. Validation and held-out test manifests are not
    opened or inspected. Foreground pixels are mask values 1 or 255; background
    pixels are 0. Intermediate mask values are reported and excluded from the
    binary pixel totals.

    Args:
        data_dir: Raw dataset root containing ``tiff/train`` and
            ``tiff/train_labels``.
        splits_dir: Directory containing the generated split manifests.
        report_path: Destination for the plain text audit summary.

    Returns:
        A dictionary containing image, corruption, mask, and pixel counts plus
        background and foreground percentages.

    Raises:
        FileNotFoundError: If the training manifest cannot be found.
    """
    train_manifest = Path(splits_dir) / "train.txt"
    if not train_manifest.is_file():
        raise FileNotFoundError(f"Training split manifest not found: {train_manifest}")
    filenames = [
        line.strip()
        for line in train_manifest.read_text(encoding="utf-8").splitlines()
        if line.strip()
    ]

    raw_dir = Path(data_dir)
    unreadable_files: set[str] = set()
    nonbinary_masks: set[str] = set()
    background_pixels = 0
    foreground_pixels = 0
    effective_background_pixels = 0
    effective_foreground_pixels = 0
    intermediate_mask_pixels = 0

    for filename in tqdm(filenames, desc="Auditing training data", unit="pair"):
        image_path, mask_path = _resolve_pair(raw_dir, filename)
        image_ok = True
        try:
            with Image.open(image_path) as image:
                image.verify()
        except (OSError, ValueError, UnidentifiedImageError):
            LOGGER.error("Unreadable or corrupted image: %s", image_path)
            unreadable_files.add(str(image_path))
            image_ok = False

        mask_array: np.ndarray | None = None
        try:
            with Image.open(mask_path) as mask:
                mask.verify()
            with Image.open(mask_path) as mask:
                mask_array = np.asarray(mask)
        except (OSError, ValueError, UnidentifiedImageError):
            LOGGER.error("Unreadable or corrupted mask: %s", mask_path)
            unreadable_files.add(str(mask_path))

        if mask_array is None:
            continue
        allowed = np.isin(mask_array, (0, 1, 255))
        if not bool(np.all(allowed)):
            nonbinary_masks.add(str(mask_path))
            LOGGER.warning("Mask contains intermediate values: %s", mask_path)

        if not image_ok:
            continue
        background_pixels += int(np.count_nonzero(mask_array == 0))
        foreground_pixels += int(np.count_nonzero(np.isin(mask_array, (1, 255))))
        effective_background_pixels += int(np.count_nonzero(mask_array == 0))
        effective_foreground_pixels += int(np.count_nonzero(mask_array > 0))
        intermediate_mask_pixels += int(np.count_nonzero(~allowed))

    binary_pixel_total = background_pixels + foreground_pixels
    if binary_pixel_total:
        background_percentage = 100.0 * background_pixels / binary_pixel_total
        foreground_percentage = 100.0 * foreground_pixels / binary_pixel_total
    else:
        background_percentage = 0.0
        foreground_percentage = 0.0
    effective_pixel_total = effective_background_pixels + effective_foreground_pixels
    if effective_pixel_total:
        effective_background_percentage = 100.0 * effective_background_pixels / effective_pixel_total
        effective_foreground_percentage = 100.0 * effective_foreground_pixels / effective_pixel_total
    else:
        effective_background_percentage = 0.0
        effective_foreground_percentage = 0.0

    results: dict[str, int | float] = {
        "total_images": len(filenames),
        "total_corrupted_files": len(unreadable_files),
        "total_nonbinary_masks": len(nonbinary_masks),
        "nonbinary_masks": len(nonbinary_masks),
        "background_pixels": background_pixels,
        "foreground_pixels": foreground_pixels,
        "intermediate_mask_pixels": intermediate_mask_pixels,
        "background_percentage": background_percentage,
        "foreground_percentage": foreground_percentage,
        "strict_background_percentage": background_percentage,
        "strict_foreground_percentage": foreground_percentage,
        "effective_background_pixels": effective_background_pixels,
        "effective_foreground_pixels": effective_foreground_pixels,
        "effective_background_percentage": effective_background_percentage,
        "effective_foreground_percentage": effective_foreground_percentage,
    }
    report = Path(report_path)
    report.parent.mkdir(parents=True, exist_ok=True)
    report.write_text(
        "Training Data Audit Summary\n"
        "============================\n"
        f"Total training images: {len(filenames)}\n"
        f"Total unreadable/corrupted files: {len(unreadable_files)}\n"
        "Pixel totals include only pairs with a readable image and mask.\n"
        f"Masks with non-binary values: {len(nonbinary_masks)}\n"
        f"Background pixels (value 0): {background_pixels}\n"
        f"Foreground pixels (value 1 or 255): {foreground_pixels}\n"
        f"Intermediate mask pixels (excluded): {intermediate_mask_pixels}\n"
        f"Strict background: {background_percentage:.12f}% of binary pixels\n"
        f"Strict foreground: {foreground_percentage:.12f}% of binary pixels\n"
        f"Training-effective background pixels (value 0): {effective_background_pixels}\n"
        f"Training-effective foreground pixels (value > 0): {effective_foreground_pixels}\n"
        f"Training-effective background: {effective_background_percentage:.12f}% of pixels\n"
        f"Training-effective foreground: {effective_foreground_percentage:.12f}% of pixels\n",
        encoding="utf-8",
    )
    LOGGER.info("Training data audit report saved to %s", report)
    return results


def main() -> None:
    """Parse command-line paths and run the training-only dataset audit."""
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--data-dir", default="data/raw", help="Raw dataset root")
    parser.add_argument("--splits-dir", default="data/splits", help="Split manifest directory")
    parser.add_argument(
        "--report-path",
        default="results/tables/data_audit_summary.txt",
        help="Output text report path",
    )
    args = parser.parse_args()
    audit_training_data(args.data_dir, args.splits_dir, args.report_path)


if __name__ == "__main__":
    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s | %(levelname)s | %(name)s | %(message)s",
    )
    main()

"""Static segmentation baselines for satellite road extraction."""

from __future__ import annotations

import torch
from torch import Tensor, nn


class ConvBlock(nn.Module):
    """Two consecutive convolution, batch-normalization, and ReLU layers."""

    def __init__(self, input_channels: int, output_channels: int) -> None:
        super().__init__()
        self.layers = nn.Sequential(
            nn.Conv2d(input_channels, output_channels, kernel_size=3, padding=1, bias=False),
            nn.BatchNorm2d(output_channels),
            nn.ReLU(inplace=True),
            nn.Conv2d(output_channels, output_channels, kernel_size=3, padding=1, bias=False),
            nn.BatchNorm2d(output_channels),
            nn.ReLU(inplace=True),
        )

    def forward(self, inputs: Tensor) -> Tensor:
        return self.layers(inputs)


class StaticUNet(nn.Module):
    """Four-stage encoder-decoder U-Net that returns one-channel logits."""

    def __init__(self, input_channels: int = 3, output_channels: int = 1) -> None:
        super().__init__()
        self.encoder1 = ConvBlock(input_channels, 64)
        self.encoder2 = ConvBlock(64, 128)
        self.encoder3 = ConvBlock(128, 256)
        self.encoder4 = ConvBlock(256, 512)
        self.pool = nn.MaxPool2d(kernel_size=2, stride=2)
        self.bottleneck = ConvBlock(512, 1024)

        self.up4 = nn.ConvTranspose2d(1024, 512, kernel_size=2, stride=2)
        self.decoder4 = ConvBlock(1024, 512)
        self.up3 = nn.ConvTranspose2d(512, 256, kernel_size=2, stride=2)
        self.decoder3 = ConvBlock(512, 256)
        self.up2 = nn.ConvTranspose2d(256, 128, kernel_size=2, stride=2)
        self.decoder2 = ConvBlock(256, 128)
        self.up1 = nn.ConvTranspose2d(128, 64, kernel_size=2, stride=2)
        self.decoder1 = ConvBlock(128, 64)
        self.output_layer = nn.Conv2d(64, output_channels, kernel_size=1)

    def forward(self, inputs: Tensor) -> Tensor:
        encoder1 = self.encoder1(inputs)
        encoder2 = self.encoder2(self.pool(encoder1))
        encoder3 = self.encoder3(self.pool(encoder2))
        encoder4 = self.encoder4(self.pool(encoder3))
        bottleneck = self.bottleneck(self.pool(encoder4))

        decoder4 = self.up4(bottleneck)
        decoder4 = self.decoder4(torch.cat((decoder4, encoder4), dim=1))
        decoder3 = self.up3(decoder4)
        decoder3 = self.decoder3(torch.cat((decoder3, encoder3), dim=1))
        decoder2 = self.up2(decoder3)
        decoder2 = self.decoder2(torch.cat((decoder2, encoder2), dim=1))
        decoder1 = self.up1(decoder2)
        decoder1 = self.decoder1(torch.cat((decoder1, encoder1), dim=1))
        return self.output_layer(decoder1)


class AttentionGate(nn.Module):
    """Learn a spatial coefficient map for filtering an encoder skip feature."""

    def __init__(self, gating_channels: int, skip_channels: int, intermediate_channels: int) -> None:
        super().__init__()
        self.weight_g = nn.Conv2d(gating_channels, intermediate_channels, kernel_size=1, bias=True)
        self.weight_x = nn.Conv2d(skip_channels, intermediate_channels, kernel_size=1, bias=True)
        self.relu = nn.ReLU(inplace=True)
        self.psi = nn.Conv2d(intermediate_channels, 1, kernel_size=1, bias=True)
        self.sigmoid = nn.Sigmoid()

    def forward(self, gating_signal: Tensor, skip_feature: Tensor) -> Tensor:
        attention_coefficients = self.sigmoid(
            self.psi(self.relu(self.weight_g(gating_signal) + self.weight_x(skip_feature)))
        )
        return skip_feature * attention_coefficients


class SoftAttentionUNet(nn.Module):
    """Four-stage U-Net that soft-attends to encoder features before fusion."""

    def __init__(self, input_channels: int = 3, output_channels: int = 1) -> None:
        super().__init__()
        self.encoder1 = ConvBlock(input_channels, 64)
        self.encoder2 = ConvBlock(64, 128)
        self.encoder3 = ConvBlock(128, 256)
        self.encoder4 = ConvBlock(256, 512)
        self.pool = nn.MaxPool2d(kernel_size=2, stride=2)
        self.bottleneck = ConvBlock(512, 1024)

        self.up4 = nn.ConvTranspose2d(1024, 512, kernel_size=2, stride=2)
        self.attention4 = AttentionGate(512, 512, 256)
        self.decoder4 = ConvBlock(1024, 512)
        self.up3 = nn.ConvTranspose2d(512, 256, kernel_size=2, stride=2)
        self.attention3 = AttentionGate(256, 256, 128)
        self.decoder3 = ConvBlock(512, 256)
        self.up2 = nn.ConvTranspose2d(256, 128, kernel_size=2, stride=2)
        self.attention2 = AttentionGate(128, 128, 64)
        self.decoder2 = ConvBlock(256, 128)
        self.up1 = nn.ConvTranspose2d(128, 64, kernel_size=2, stride=2)
        self.attention1 = AttentionGate(64, 64, 32)
        self.decoder1 = ConvBlock(128, 64)
        self.output_layer = nn.Conv2d(64, output_channels, kernel_size=1)

    def forward(self, inputs: Tensor) -> Tensor:
        encoder1 = self.encoder1(inputs)
        encoder2 = self.encoder2(self.pool(encoder1))
        encoder3 = self.encoder3(self.pool(encoder2))
        encoder4 = self.encoder4(self.pool(encoder3))
        bottleneck = self.bottleneck(self.pool(encoder4))

        decoder4 = self.up4(bottleneck)
        attended4 = self.attention4(decoder4, encoder4)
        decoder4 = self.decoder4(torch.cat((decoder4, attended4), dim=1))
        decoder3 = self.up3(decoder4)
        attended3 = self.attention3(decoder3, encoder3)
        decoder3 = self.decoder3(torch.cat((decoder3, attended3), dim=1))
        decoder2 = self.up2(decoder3)
        attended2 = self.attention2(decoder2, encoder2)
        decoder2 = self.decoder2(torch.cat((decoder2, attended2), dim=1))
        decoder1 = self.up1(decoder2)
        attended1 = self.attention1(decoder1, encoder1)
        decoder1 = self.decoder1(torch.cat((decoder1, attended1), dim=1))
        return self.output_layer(decoder1)

===== src/dataloaders.py =====
"""Dataset and DataLoader construction for leakage-free road segmentation."""

from __future__ import annotations

from pathlib import Path
from typing import Callable

import numpy as np
import torch
from PIL import Image
from torch import Tensor
from torch.utils.data import DataLoader, Dataset

from src.transforms import get_train_transforms, get_val_test_transforms

PairTransform = Callable[[Image.Image, Image.Image], tuple[Tensor, Tensor]]
IMAGE_SUFFIXES = {".tif", ".tiff", ".png", ".jpg", ".jpeg"}
RANDOM_SEED = 67


class MassachusettsRoadsDataset(Dataset[tuple[Tensor, Tensor]]):
    """Load RGB image and binary road-mask pairs listed in a split manifest.

    Manifest entries contain image filenames. Images are read from
    ``raw_data_root/tiff/train`` and masks are matched by stem in
    ``raw_data_root/tiff/train_labels``.
    """

    def __init__(
        self,
        split_manifest: str | Path,
        raw_data_root: str | Path,
        transforms: PairTransform | None = None,
    ) -> None:
        """Initialize manifest entries, raw paths, and an optional pair transform."""
        self.split_manifest = Path(split_manifest)
        self.raw_data_root = Path(raw_data_root)
        self.image_dir = self.raw_data_root / "tiff" / "train"
        self.mask_dir = self.raw_data_root / "tiff" / "train_labels"
        self.transforms = transforms

        if not self.split_manifest.is_file():
            raise FileNotFoundError(f"Split manifest not found: {self.split_manifest}")
        if not self.image_dir.is_dir():
            raise FileNotFoundError(f"Image directory not found: {self.image_dir}")
        if not self.mask_dir.is_dir():
            raise FileNotFoundError(f"Mask directory not found: {self.mask_dir}")

        self.filenames = [
            line.strip()
            for line in self.split_manifest.read_text(encoding="utf-8").splitlines()
            if line.strip()
        ]
        masks_by_stem: dict[str, Path] = {}
        for path in self.mask_dir.iterdir():
            if path.is_file() and path.suffix.casefold() in IMAGE_SUFFIXES:
                key = path.stem.casefold()
                if key in masks_by_stem:
                    raise ValueError(f"Multiple masks share the same filename stem: {path.stem}")
                masks_by_stem[key] = path

        self.samples: list[tuple[Path, Path]] = []
        for filename in self.filenames:
            image_path = self.image_dir / filename
            mask_path = masks_by_stem.get(Path(filename).stem.casefold())
            if not image_path.is_file():
                raise FileNotFoundError(f"Image listed in manifest not found: {image_path}")
            if mask_path is None:
                raise FileNotFoundError(f"No mask found for image listed in manifest: {filename}")
            self.samples.append((image_path, mask_path))

    def __len__(self) -> int:
        """Return the number of image-mask pairs in this manifest."""
        return len(self.samples)

    def __getitem__(self, index: int) -> tuple[Tensor, Tensor]:
        """Load, transform, and return one normalized image/mask tensor pair."""
        image_path, mask_path = self.samples[index]
        with Image.open(image_path) as image_file:
            image = image_file.convert("RGB")
        with Image.open(mask_path) as mask_file:
            mask = mask_file.convert("L")

        if self.transforms is not None:
            image_tensor, mask_tensor = self.transforms(image, mask)
        else:
            image_array = np.asarray(image, dtype=np.uint8).copy()
            mask_array = np.asarray(mask, dtype=np.uint8).copy()
            image_tensor = torch.from_numpy(image_array).permute(2, 0, 1).to(torch.float32).div_(255.0)
            mask_tensor = torch.from_numpy(mask_array).unsqueeze(0).to(torch.float32)
            mask_tensor = (mask_tensor > 0).to(torch.float32)

        if image_tensor.ndim != 3 or image_tensor.shape[0] != 3:
            raise ValueError(f"Expected image tensor [3,H,W], got {tuple(image_tensor.shape)}")
        if mask_tensor.ndim == 2:
            mask_tensor = mask_tensor.unsqueeze(0)
        if mask_tensor.ndim != 3 or mask_tensor.shape[0] != 1:
            raise ValueError(f"Expected mask tensor [1,H,W], got {tuple(mask_tensor.shape)}")
        mask_tensor = (mask_tensor > 0.5).to(torch.float32)
        return image_tensor.to(torch.float32), mask_tensor


def get_dataloaders(
    batch_size: int,
    num_workers: int = 4,
    data_dir: str | Path = "data/raw",
    splits_dir: str | Path = "data/splits",
) -> tuple[DataLoader[tuple[Tensor, Tensor]], DataLoader[tuple[Tensor, Tensor]], DataLoader[tuple[Tensor, Tensor]]]:
    """Create train, validation, and test loaders from the three manifests.

    The training loader shuffles with a fixed generator seed. Validation and
    test loaders preserve manifest order. All three use only their own manifest
    and receive the appropriate paired transforms.
    """
    if batch_size <= 0:
        raise ValueError("batch_size must be a positive integer")
    if num_workers < 0:
        raise ValueError("num_workers must be nonnegative")

    split_dir = Path(splits_dir)
    raw_data_root = Path(data_dir)
    train_dataset = MassachusettsRoadsDataset(
        split_dir / "train.txt", raw_data_root, get_train_transforms()
    )
    validation_dataset = MassachusettsRoadsDataset(
        split_dir / "val.txt", raw_data_root, get_val_test_transforms()
    )
    test_dataset = MassachusettsRoadsDataset(
        split_dir / "test.txt", raw_data_root, get_val_test_transforms()
    )

    generator = torch.Generator()
    generator.manual_seed(RANDOM_SEED)
    common_loader_options = {
        "batch_size": batch_size,
        "num_workers": num_workers,
        "pin_memory": torch.cuda.is_available(),
        "persistent_workers": num_workers > 0,
    }
    train_loader = DataLoader(
        train_dataset, shuffle=True, generator=generator, **common_loader_options
    )
    validation_loader = DataLoader(
        validation_dataset, shuffle=False, **common_loader_options
    )
    test_loader = DataLoader(test_dataset, shuffle=False, **common_loader_options)
    return train_loader, validation_loader, test_loader

===== src/dataset.py =====
"""Create reproducible, leakage-aware filename splits for the roads dataset."""

from __future__ import annotations

import logging
import random
from pathlib import Path

LOGGER = logging.getLogger(__name__)
RANDOM_SEED = 67
IMAGE_DIRECTORY = Path("tiff") / "train"
MASK_DIRECTORY = Path("tiff") / "train_labels"
IMAGE_SUFFIXES = {".tif", ".tiff", ".png", ".jpg", ".jpeg"}


def create_deterministic_splits(data_dir: str, splits_dir: str) -> None:
    """Find paired image/mask files and write seeded 70/15/15 split manifests.

    ``data_dir`` is the raw data directory (normally ``data/raw``). Each
    manifest stores image filenames, one per line; masks are matched by stem
    under ``tiff/train_labels``. Pair matching is case-insensitive by stem.

    Args:
        data_dir: Root of the raw dataset directory.
        splits_dir: Directory in which train, validation, and test manifests
            will be written.

    Raises:
        FileNotFoundError: If either expected source directory is missing.
        ValueError: If no valid image/mask pairs are found.
    """
    raw_root = Path(data_dir)
    image_dir = raw_root / IMAGE_DIRECTORY
    mask_dir = raw_root / MASK_DIRECTORY
    if not image_dir.is_dir():
        raise FileNotFoundError(f"Image directory not found: {image_dir}")
    if not mask_dir.is_dir():
        raise FileNotFoundError(f"Mask directory not found: {mask_dir}")

    masks_by_stem = {
        path.stem.casefold(): path
        for path in mask_dir.iterdir()
        if path.is_file() and path.suffix.casefold() in IMAGE_SUFFIXES
    }
    pairs = []
    for image_path in sorted(image_dir.iterdir(), key=lambda path: path.name.casefold()):
        if image_path.is_file() and image_path.suffix.casefold() in IMAGE_SUFFIXES:
            mask_path = masks_by_stem.get(image_path.stem.casefold())
            if mask_path is not None:
                pairs.append(image_path.name)
            else:
                LOGGER.warning("Skipping image without matching mask: %s", image_path)

    if not pairs:
        raise ValueError(f"No valid image-mask pairs found in {image_dir} and {mask_dir}")

    random.Random(RANDOM_SEED).shuffle(pairs)
    train_end = int(len(pairs) * 0.70)
    validation_end = train_end + int(len(pairs) * 0.15)
    partitions = {
        "train": pairs[:train_end],
        "val": pairs[train_end:validation_end],
        "test": pairs[validation_end:],
    }

    output_dir = Path(splits_dir)
    output_dir.mkdir(parents=True, exist_ok=True)
    for split_name, filenames in partitions.items():
        manifest = output_dir / f"{split_name}.txt"
        manifest.write_text("".join(f"{filename}\n" for filename in filenames), encoding="utf-8")
        LOGGER.info("Wrote %d filenames to %s", len(filenames), manifest)


def main() -> None:
    """Regenerate split manifests with the configured dataset paths."""
    import argparse

    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--data-dir", default="data/raw", help="Raw dataset root")
    parser.add_argument("--splits-dir", default="data/splits", help="Split manifest directory")
    args = parser.parse_args()
    logging.basicConfig(level=logging.INFO, format="%(levelname)s: %(message)s")
    create_deterministic_splits(args.data_dir, args.splits_dir)


if __name__ == "__main__":
    main()

===== src/dynamic_unet.py =====
"""Dynamic U-Net with entropy-gated skip pruning and dual-path decoding."""

from __future__ import annotations

import torch
import torch.nn.functional as functional
from torch import Tensor, nn

from src.entropy_gate import ShannonEntropyGate


class EncoderBlock(nn.Module):
    """Two convolutional layers with batch normalization and ReLU activation."""

    def __init__(self, input_channels: int, output_channels: int) -> None:
        super().__init__()
        self.layers = nn.Sequential(
            nn.Conv2d(input_channels, output_channels, kernel_size=3, padding=1, bias=False),
            nn.BatchNorm2d(output_channels),
            nn.ReLU(inplace=True),
            nn.Conv2d(output_channels, output_channels, kernel_size=3, padding=1, bias=False),
            nn.BatchNorm2d(output_channels),
            nn.ReLU(inplace=True),
        )

    def forward(self, feature_map: Tensor) -> Tensor:
        return self.layers(feature_map)


class DualPathDecoderBlock(nn.Module):
    """Route active sites through sparse 3x3 layers and others through 1x1."""

    def __init__(self, channels: int) -> None:
        super().__init__()
        self.high_entropy_convolution1 = nn.Conv2d(
            channels, channels, kernel_size=3, padding=1, bias=False
        )
        self.high_entropy_normalization1 = nn.LayerNorm(channels)
        self.high_entropy_convolution2 = nn.Conv2d(
            channels, channels, kernel_size=3, padding=1, bias=False
        )
        self.high_entropy_normalization2 = nn.LayerNorm(channels)
        self.low_entropy_projection = nn.Conv2d(
            channels, channels, kernel_size=1, bias=False
        )
        self.low_entropy_normalization = nn.LayerNorm(channels)
        self.activation = nn.ReLU(inplace=True)

    @staticmethod
    def _sparse_convolution(
        feature_map: Tensor,
        convolution: nn.Conv2d,
        active_mask: Tensor,
    ) -> Tensor:
        batch_size, _, height, width = feature_map.shape
        unfolded_patches = functional.unfold(
            feature_map,
            kernel_size=convolution.kernel_size,
            dilation=convolution.dilation,
            padding=convolution.padding,
            stride=convolution.stride,
        )
        flattened_outputs = []
        for batch_index in range(batch_size):
            active_indices = active_mask[batch_index, 0].reshape(-1).nonzero(as_tuple=False).reshape(-1)
            if active_indices.numel() == 0:
                flattened_output = feature_map.new_zeros(
                    (convolution.out_channels, height * width)
                )
            else:
                active_patches = unfolded_patches[batch_index].index_select(
                    1, active_indices
                ).transpose(0, 1)
                active_values = functional.linear(
                    active_patches,
                    convolution.weight.flatten(start_dim=1),
                    convolution.bias,
                ).transpose(0, 1)
                flattened_output = feature_map.new_zeros(
                    (convolution.out_channels, height * width)
                ).index_copy(1, active_indices, active_values.to(feature_map.dtype))
            flattened_outputs.append(
                flattened_output.reshape(convolution.out_channels, height, width)
            )
        return torch.stack(flattened_outputs, dim=0)

    @staticmethod
    def _normalize_channels(feature_map: Tensor, normalization: nn.LayerNorm) -> Tensor:
        channels_last = feature_map.permute(0, 2, 3, 1)
        return normalization(channels_last).permute(0, 3, 1, 2)

    def forward(self, fused_feature: Tensor, routing_mask: Tensor) -> Tensor:
        active_mask = routing_mask.to(dtype=torch.bool)
        active_neighborhood = functional.max_pool2d(
            active_mask.to(dtype=fused_feature.dtype), kernel_size=3, stride=1, padding=1
        ).to(dtype=torch.bool)
        high_entropy_features = self._sparse_convolution(
            fused_feature,
            self.high_entropy_convolution1,
            active_neighborhood,
        )
        high_entropy_features = self.activation(
            self._normalize_channels(
                high_entropy_features,
                self.high_entropy_normalization1,
            )
        )
        high_entropy_features = torch.where(
            active_neighborhood,
            high_entropy_features,
            torch.zeros_like(high_entropy_features),
        )
        high_entropy_features = self._sparse_convolution(
            high_entropy_features,
            self.high_entropy_convolution2,
            active_mask,
        )
        high_entropy_features = self.activation(
            self._normalize_channels(
                high_entropy_features,
                self.high_entropy_normalization2,
            )
        )
        low_entropy_features = self.low_entropy_projection(fused_feature)
        low_entropy_features = self.activation(
            self._normalize_channels(
                low_entropy_features,
                self.low_entropy_normalization,
            )
        )
        return torch.where(
            active_mask,
            fused_feature + high_entropy_features,
            low_entropy_features,
        )


class DenseDecoderStage(nn.Module):
    """Upsample and densely fuse the coarsest decoder skip feature."""

    def __init__(self, input_channels: int, skip_channels: int, output_channels: int) -> None:
        super().__init__()
        self.upsample = nn.ConvTranspose2d(
            input_channels, output_channels, kernel_size=2, stride=2
        )
        self.fusion = EncoderBlock(output_channels + skip_channels, output_channels)

    def forward(self, decoder_feature: Tensor, skip_feature: Tensor) -> Tensor:
        upsampled_feature = self.upsample(decoder_feature)
        return self.fusion(torch.cat((upsampled_feature, skip_feature), dim=1))


class GatedDecoderStage(nn.Module):
    """Predict entropy, prune a skip, fuse features, and execute both lanes."""

    def __init__(
        self,
        input_channels: int,
        skip_channels: int,
        output_channels: int,
        threshold: float,
    ) -> None:
        super().__init__()
        self.upsample = nn.ConvTranspose2d(
            input_channels, output_channels, kernel_size=2, stride=2
        )
        self.auxiliary_head = nn.Conv2d(output_channels, 1, kernel_size=1)
        self.entropy_gate = ShannonEntropyGate(threshold=threshold)
        self.fusion = nn.Sequential(
            nn.Conv2d(output_channels + skip_channels, output_channels, kernel_size=1, bias=False),
            nn.BatchNorm2d(output_channels),
            nn.ReLU(inplace=True),
        )
        self.decoder_block = DualPathDecoderBlock(output_channels)

    def forward(
        self,
        decoder_feature: Tensor,
        skip_feature: Tensor,
    ) -> tuple[Tensor, Tensor, Tensor]:
        upsampled_feature = self.upsample(decoder_feature)
        auxiliary_logits = self.auxiliary_head(upsampled_feature)
        routing_mask, _ = self.entropy_gate(auxiliary_logits)
        pruned_skip = skip_feature * routing_mask.to(dtype=skip_feature.dtype)
        fused_feature = self.fusion(
            torch.cat((upsampled_feature, pruned_skip), dim=1)
        )
        routed_feature = self.decoder_block(fused_feature, routing_mask)
        return routed_feature, auxiliary_logits, routing_mask


class DynamicUNet(nn.Module):
    """Four-stage U-Net with deep-supervised entropy-routed fine decoders."""

    def __init__(
        self,
        input_channels: int = 3,
        output_channels: int = 1,
        threshold: float = 0.5,
    ) -> None:
        super().__init__()
        self.threshold = float(threshold)
        self.encoder1 = EncoderBlock(input_channels, 64)
        self.encoder2 = EncoderBlock(64, 128)
        self.encoder3 = EncoderBlock(128, 256)
        self.encoder4 = EncoderBlock(256, 512)
        self.pool = nn.MaxPool2d(kernel_size=2, stride=2)
        self.bottleneck = EncoderBlock(512, 1024)
        self.decoder4 = DenseDecoderStage(1024, 512, 512)
        self.decoder3 = GatedDecoderStage(512, 256, 256, self.threshold)
        self.decoder2 = GatedDecoderStage(256, 128, 128, self.threshold)
        self.decoder1 = GatedDecoderStage(128, 64, 64, self.threshold)
        self.output_layer = nn.Conv2d(64, output_channels, kernel_size=1)

    def set_entropy_threshold(self, threshold: float) -> None:
        """Update the threshold used by every internal Shannon entropy gate."""
        if not 0.0 <= threshold <= 1.0:
            raise ValueError("Entropy threshold must be in the range [0, 1]")
        self.threshold = float(threshold)
        for module in self.modules():
            if isinstance(module, ShannonEntropyGate):
                module.threshold = self.threshold

    def forward(
        self,
        inputs: Tensor,
        return_internals: bool = False,
    ) -> Tensor | tuple[Tensor, list[Tensor], list[Tensor]]:
        encoder1 = self.encoder1(inputs)
        encoder2 = self.encoder2(self.pool(encoder1))
        encoder3 = self.encoder3(self.pool(encoder2))
        encoder4 = self.encoder4(self.pool(encoder3))
        bottleneck = self.bottleneck(self.pool(encoder4))
        decoder4 = self.decoder4(bottleneck, encoder4)
        decoder3, auxiliary_logits3, routing_mask3 = self.decoder3(decoder4, encoder3)
        decoder2, auxiliary_logits2, routing_mask2 = self.decoder2(decoder3, encoder2)
        decoder1, auxiliary_logits1, routing_mask1 = self.decoder1(decoder2, encoder1)
        logits = self.output_layer(decoder1)
        if return_internals:
            return (
                logits,
                [auxiliary_logits3, auxiliary_logits2, auxiliary_logits1],
                [routing_mask3, routing_mask2, routing_mask1],
            )
        return logits

import logging

from typing import Sequence

matplotlib.use("Agg")
import matplotlib.pyplot as plt

SEED = 67

VALID_IMAGE_SUFFIXES = {".tif", ".tiff", ".png", ".jpg", ".jpeg"}


def _read_training_filenames(split_path: Path) -> list[str]:
    """Read only the training manifest and return its nonempty filenames."""
    if not split_path.is_file():
        raise FileNotFoundError(f"Training split manifest not found: {split_path}")
    return [line.strip() for line in split_path.read_text(encoding="utf-8").splitlines() if line.strip()]


def _load_rgb_tensor(image_path: Path, device: torch.device) -> torch.Tensor:
    """Load an image as an RGB float tensor in CHW layout on ``device``."""
    with Image.open(image_path) as image:
        rgb = np.asarray(image.convert("RGB"), dtype=np.uint8).copy()
    return torch.from_numpy(rgb).permute(2, 0, 1).to(device=device, dtype=torch.float32).div_(255.0)


def _load_binary_mask(mask_path: Path, device: torch.device) -> torch.Tensor:
    """Load a road mask as a 1x1xHxW float tensor on ``device``."""
    with Image.open(mask_path) as mask:
        values = np.asarray(mask).copy()
    if values.ndim == 3:
        values = values[..., 0]
    binary = (values > 0).astype(np.float32, copy=False)
    return torch.from_numpy(binary).unsqueeze(0).unsqueeze(0).to(device=device)


def _resolve_mask_path(mask_dir: Path, filename: str) -> Path:
    """Find a mask by case-insensitive stem across supported image suffixes."""
    stem = Path(filename).stem.casefold()
    for candidate in mask_dir.iterdir():
        if candidate.is_file() and candidate.suffix.casefold() in VALID_IMAGE_SUFFIXES:
            if candidate.stem.casefold() == stem:
                return candidate
    raise FileNotFoundError(f"No mask found for {filename} in {mask_dir}")


def _training_class_percentages(
    filenames: Sequence[str], mask_dir: Path
) -> tuple[float, float]:
    """Count exact training-manifest background and effective foreground pixels."""
    background_pixels = 0
    foreground_pixels = 0
    for filename in filenames:
        with Image.open(_resolve_mask_path(mask_dir, filename)) as mask:
            values = np.asarray(mask.convert("L"))
        foreground_pixels += int(np.count_nonzero(values > 0))
        background_pixels += int(np.count_nonzero(values == 0))
    total_pixels = background_pixels + foreground_pixels
    if total_pixels == 0:
        return 95.24, 4.76
    return 100.0 * background_pixels / total_pixels, 100.0 * foreground_pixels / total_pixels


def _plot_class_imbalance(

def _calculate_rgb_statistics(
    filenames: Sequence[str], image_dir: Path, output_dir: Path, device: torch.device
) -> tuple[torch.Tensor, torch.Tensor]:
    """Aggregate global RGB mean, population standard deviation, and histograms.

    Image tensors and all channel reductions/histogram accumulation run on the
    selected torch device. Only the final small histogram arrays are copied to
    CPU for Matplotlib rendering.
    """
    channel_sum = torch.zeros(3, dtype=torch.float64, device=device)
    channel_square_sum = torch.zeros(3, dtype=torch.float64, device=device)
    channel_histograms = torch.zeros((3, 256), dtype=torch.float64, device=device)
    pixel_count = 0

    for filename in filenames:
        image_tensor = _load_rgb_tensor(image_dir / filename, device)
        image64 = image_tensor.to(dtype=torch.float64)
        channel_sum += image64.sum(dim=(1, 2))
        channel_square_sum += image64.square().sum(dim=(1, 2))
        for channel_index in range(3):
            channel_histograms[channel_index] += torch.histc(
                image_tensor[channel_index], bins=256, min=0.0, max=1.0
            ).to(dtype=torch.float64)
        pixel_count += image_tensor.shape[1] * image_tensor.shape[2]

    if pixel_count == 0:
        raise ValueError("The training manifest contains no images")
    means = channel_sum / pixel_count
    variances = torch.clamp(channel_square_sum / pixel_count - means.square(), min=0.0)
    standard_deviations = variances.sqrt()

    LOGGER.info(
        "Training-fold RGB mean: R=%.8f, G=%.8f, B=%.8f",
        *means.detach().cpu().tolist(),
    )
    statistics_path = Path("results/tables/rgb_normalization_stats.json")
    statistics_path.parent.mkdir(parents=True, exist_ok=True)
    statistics_path.write_text(
        json.dumps(
            {
                "mean": means.detach().cpu().tolist(),
                "std": standard_deviations.detach().cpu().tolist(),
            },
            indent=2,
        )
        + "\n",
        encoding="utf-8",
    )
    LOGGER.info(
        "Training-fold RGB std:  R=%.8f, G=%.8f, B=%.8f",
        *standard_deviations.detach().cpu().tolist(),
    )

    histogram_cpu = channel_histograms.detach().cpu().numpy()
    intensity_centers = np.linspace(0.0, 1.0, 256)
    figure, axis = plt.subplots(figsize=(9, 5), constrained_layout=True)
    for channel_index, (channel_name, color) in enumerate(
        zip(CHANNEL_NAMES, ("red", "green", "blue"))
    ):
        axis.plot(
            intensity_centers,
            histogram_cpu[channel_index],
            color=color,
            alpha=0.78,
            linewidth=1.25,
            label=channel_name,
        )
    axis.set_xlabel("Normalized pixel intensity")
    axis.set_ylabel("Pixel count")
    axis.set_title("Training-fold RGB Pixel Distribution")
    axis.legend()
    axis.grid(alpha=0.2)
    figure.savefig(output_dir / "rgb_distribution.png", dpi=300, bbox_inches="tight")
    plt.close(figure)
    return means, standard_deviations

===== src/entropy_gate.py =====
"""Parameter-free Shannon entropy routing with a straight-through estimator."""

from __future__ import annotations

import torch
from torch import Tensor, nn


class StraightThroughEstimator(torch.autograd.Function):
    """Binarize continuous entropy while passing upstream gradients unchanged."""

    @staticmethod
    def forward(ctx: object, entropy: Tensor, threshold: float) -> Tensor:
        return (entropy > threshold).to(dtype=entropy.dtype)

    @staticmethod
    def backward(ctx: object, output_gradient: Tensor) -> tuple[Tensor, None]:
        return output_gradient, None


class ShannonEntropyGate(nn.Module):
    """Threshold normalized binary entropy from one-channel auxiliary logits."""

    def __init__(self, threshold: float = 0.5, eps: float = 1e-7) -> None:
        super().__init__()
        self.threshold = float(threshold)
        self.eps = float(eps)

    def forward(self, auxiliary_logits: Tensor) -> tuple[Tensor, Tensor]:
        probabilities = torch.sigmoid(auxiliary_logits.float()).clamp(
            min=self.eps,
            max=1.0 - self.eps,
        )
        entropy = -(
            probabilities * torch.log2(probabilities)
            + (1.0 - probabilities) * torch.log2(1.0 - probabilities)
        )
        routing_mask = StraightThroughEstimator.apply(entropy, self.threshold)
        return routing_mask, entropy

"""Reusable inference metrics, threshold sweeps, and routing policy studies."""

from __future__ import annotations

from collections.abc import Sequence
from typing import Any

import torch

from torch import Tensor

from src.dynamic_unet import DynamicUNet

try:
    from scipy.ndimage import distance_transform_edt
except ImportError:
    distance_transform_edt = None


def extract_binary_boundary(binary_mask: Tensor, radius: int = 1) -> Tensor:
    """Extract a binary morphological-gradient boundary using replicate padding."""
    if radius < 0:
        raise ValueError("radius must be nonnegative")
    binary_mask = binary_mask.float()
    if radius == 0:
        return torch.zeros_like(binary_mask, dtype=torch.bool)
    kernel_size = 2 * radius + 1
    padded_mask = functional.pad(
        binary_mask, (radius, radius, radius, radius), mode="replicate"
    )
    dilated_mask = functional.max_pool2d(padded_mask, kernel_size, stride=1)
    eroded_mask = -functional.max_pool2d(-padded_mask, kernel_size, stride=1)
    return dilated_mask > eroded_mask


def compute_boundary_f1(
    logits: Tensor,
    targets: Tensor,
    boundary_radius: int = 1,
    tolerance_radius: int = 2,
) -> float:
    """Compute symmetric boundary F1 with a configurable pixel matching band."""
    predicted_boundary = extract_binary_boundary(torch.sigmoid(logits) >= 0.5, boundary_radius)
    target_boundary = extract_binary_boundary(targets >= 0.5, boundary_radius)
    predicted_count = int(predicted_boundary.sum().item())
    target_count = int(target_boundary.sum().item())
    if predicted_count == 0 and target_count == 0:
        return 1.0
    if predicted_count == 0 or target_count == 0:
        return 0.0
    kernel_size = 2 * tolerance_radius + 1
    predicted_neighborhood = functional.max_pool2d(
        predicted_boundary.float(), kernel_size, stride=1, padding=tolerance_radius
    ) > 0
    target_neighborhood = functional.max_pool2d(
        target_boundary.float(), kernel_size, stride=1, padding=tolerance_radius
    ) > 0
    matched_predictions = int((predicted_boundary & target_neighborhood).sum().item())
    matched_targets = int((target_boundary & predicted_neighborhood).sum().item())
    precision = matched_predictions / predicted_count
    recall = matched_targets / target_count
    return 2.0 * precision * recall / (precision + recall) if precision + recall else 0.0


def _directed_distances_torch(
    source_coordinates: Tensor, destination_coordinates: Tensor, chunk_size: int = 1024
) -> Tensor:
    """Compute chunked nearest-neighbor distances without a full pairwise matrix."""
    distance_chunks = []
    for chunk_start in range(0, source_coordinates.shape[0], chunk_size):
        coordinate_chunk = source_coordinates[chunk_start : chunk_start + chunk_size]
        distance_chunks.append(torch.cdist(coordinate_chunk, destination_coordinates).min(dim=1).values)
    return torch.cat(distance_chunks)


def compute_hd95(
    logits: Tensor, targets: Tensor, boundary_radius: int = 1
) -> tuple[float, bool]:
    """Compute symmetric 95th-percentile boundary distance and validity flag.

    The boolean is true only when both predicted and target boundaries are
    nonempty. Empty-set cases receive a finite diagonal-distance penalty (or
    zero when both are empty) and are excluded from valid HD95 summaries.
    """
    predicted_boundary = extract_binary_boundary(
        torch.sigmoid(logits) >= 0.5, boundary_radius
    )
    target_boundary = extract_binary_boundary(targets >= 0.5, boundary_radius)
    per_image_distances: list[float] = []
    all_valid = True
    for predicted_mask, target_mask in zip(predicted_boundary, target_boundary):
        predicted_coordinates = torch.nonzero(predicted_mask[0], as_tuple=False).float()
        target_coordinates = torch.nonzero(target_mask[0], as_tuple=False).float()
        if not predicted_coordinates.numel() and not target_coordinates.numel():
            per_image_distances.append(0.0)
            all_valid = False
            continue
        if not predicted_coordinates.numel() or not target_coordinates.numel():
            height, width = predicted_mask.shape[-2:]
            per_image_distances.append(float((height**2 + width**2) ** 0.5))
            all_valid = False
            continue
        if distance_transform_edt is not None:
            predicted_array = predicted_mask[0].detach().cpu().numpy().astype(bool)
            target_array = target_mask[0].detach().cpu().numpy().astype(bool)
            predicted_to_target = distance_transform_edt(~target_array)[predicted_array]
            target_to_predicted = distance_transform_edt(~predicted_array)[target_array]
            distances = np.concatenate((predicted_to_target, target_to_predicted))
            per_image_distances.append(float(np.percentile(distances, 95)))
        else:
            predicted_to_target = _directed_distances_torch(
                predicted_coordinates, target_coordinates
            )
            target_to_predicted = _directed_distances_torch(
                target_coordinates, predicted_coordinates
            )
            distances = torch.cat((predicted_to_target, target_to_predicted))
            per_image_distances.append(float(torch.quantile(distances, 0.95).item()))
    return (float(np.mean(per_image_distances)) if per_image_distances else 0.0, all_valid)


def bootstrap_confidence_interval(
    values: Sequence[float],
    num_bootstrap: int = 1000,
    confidence: float = 0.95,
    seed: int = 67,
) -> tuple[float, float, float]:
    """Return the sample mean and seeded percentile bootstrap confidence bounds."""
    sample = np.asarray(values, dtype=np.float64)
    if sample.size == 0:
        return float("nan"), float("nan"), float("nan")
    if num_bootstrap <= 0:
        raise ValueError("num_bootstrap must be positive")
    if not 0.0 < confidence < 1.0:
        raise ValueError("confidence must be between 0 and 1")
    generator = np.random.default_rng(seed)
    indices = generator.integers(0, sample.size, size=(num_bootstrap, sample.size))
    means = sample[indices].mean(axis=1)
    tail = (1.0 - confidence) / 2.0
    lower, upper = np.quantile(means, (tail, 1.0 - tail))
    return float(sample.mean()), float(lower), float(upper)


def _update_counts(total: dict[str, int], batch: dict[str, int]) -> None:
    """Accumulate the four confusion count fields in place."""
    for key, value in batch.items():
        total[key] = total.get(key, 0) + value


def _mask_counts(
    routing_masks: Sequence[Tensor],
) -> tuple[list[float], list[int], list[float], list[int]]:
    """Return raw and dilated active counts and pixel totals per stage."""
    raw_counts = [float(mask.float().sum().item()) for mask in routing_masks]
    raw_totals = [mask.numel() for mask in routing_masks]
    dilated_masks = [compute_dilated_routing_mask(mask) for mask in routing_masks]
    dilated_counts = [float(mask.float().sum().item()) for mask in dilated_masks]
    dilated_totals = [mask.numel() for mask in dilated_masks]
    return raw_counts, raw_totals, dilated_counts, dilated_totals


@torch.no_grad()
def evaluate_threshold_sweep(

    device = torch.device(device)

    original_training_state = model.training
    results: list[dict[str, float]] = []
    try:

                raw_counts, raw_totals, halo_counts, halo_totals = _mask_counts(routing_masks)

                    halo_stage_totals[stage_index] += halo_totals[stage_index]
            raw_fractions = [
                count / total if total else 0.0
                for count, total in zip(raw_stage_counts or [], raw_stage_totals or [])
            ]
            halo_fractions = [
                count / total if total else 0.0

    return results


def _policy_hook(stage_index: int, state: dict[str, Any], seed: int):
    """Build a per-sample random/inverted route override hook for one gate."""
    def apply_policy(module: torch.nn.Module, module_inputs: tuple[Any, ...], module_outputs: Any):
        """Replace gate masks with per-sample budget-matched alternatives."""
        routing_mask, entropy_map = module_outputs
        policy = state["policy"]
        if policy not in ("random", "inverted"):
            return module_outputs
        replacement_mask = torch.zeros_like(routing_mask)
        batch_size, _, height, width = routing_mask.shape
        for sample_index in range(batch_size):
            active_count = int(routing_mask[sample_index, 0].sum().item())
            flattened_mask = replacement_mask[sample_index, 0].reshape(-1)
            sample_counter = state["sample_counters"][stage_index]
            state["sample_counters"][stage_index] += 1
            if active_count == 0:
                continue
            if policy == "random":
                generator = torch.Generator(device=routing_mask.device)
                generator.manual_seed(
                    seed + sample_counter * 31 + stage_index
                )
                selected_indices = torch.randperm(
                    height * width, generator=generator, device=routing_mask.device
                )[:active_count]
            else:
                flattened_entropy = entropy_map[sample_index, 0].reshape(-1)
                selected_indices = torch.topk(
                    flattened_entropy, k=active_count, largest=False
                ).indices
            flattened_mask[selected_indices] = 1.0
        return replacement_mask, entropy_map

    return apply_policy

===== src/flops_profiler.py =====
"""Analytical convolution FLOPs and model parameter accounting utilities."""

from __future__ import annotations

from collections.abc import Sequence

import torch.nn.functional as functional
from torch import Tensor, nn


def convolution_gflops(
    spatial_size: int, input_channels: int, output_channels: int, kernel_size: int
) -> float:
    """Return multiply-add GFLOPs for a same-resolution 2D convolution."""
    return (
        2.0 * spatial_size**2 * input_channels * output_channels * kernel_size**2
        / 1e9
    )


def decoder_transpose_gflops(
    spatial_size: int,
    input_channels: int,
    output_channels: int,
    kernel_size: int = 2,
) -> float:
    """Return multiply-add GFLOPs for stride-two transposed convolution."""
    input_spatial_size = spatial_size // 2
    return (
        2.0 * input_spatial_size**2 * input_channels * output_channels * kernel_size**2
        / 1e9
    )


def encoder_bottleneck_gflops(
    input_resolution: int = 512, input_channels: int = 3
) -> float:
    """Return two-convolution FLOPs for the shared encoder and bottleneck."""
    total_cost = 0.0
    stage_specs = ((64, input_channels), (128, 64), (256, 128), (512, 256))
    for stage_index, (output_channels, stage_input_channels) in enumerate(stage_specs):
        spatial_size = input_resolution // (2**stage_index)
        total_cost += convolution_gflops(
            spatial_size, stage_input_channels, output_channels, 3
        )
        total_cost += convolution_gflops(
            spatial_size, output_channels, output_channels, 3
        )
    bottleneck_resolution = input_resolution // 16
    total_cost += convolution_gflops(
        bottleneck_resolution, 512, 1024, 3
    )
    total_cost += convolution_gflops(
        bottleneck_resolution, 1024, 1024, 3
    )
    return total_cost


_DECODER_STAGE_SPECS = (
    (64, 1024, 512),
    (128, 512, 256),
    (256, 256, 128),
    (512, 128, 64),
)
_GATED_STAGE_SPECS = _DECODER_STAGE_SPECS[1:]


def static_decoder_gflops(input_resolution: int = 512) -> float:
    """Return analytical dense decoder FLOPs for the four-stage Static U-Net."""
    total_cost = 0.0
    for spatial_size, input_channels, output_channels in _DECODER_STAGE_SPECS:
        scaled_spatial_size = spatial_size * input_resolution // 512
        total_cost += decoder_transpose_gflops(
            scaled_spatial_size, input_channels, output_channels
        )
        total_cost += convolution_gflops(
            scaled_spatial_size, output_channels * 2, output_channels, 3
        )
        total_cost += convolution_gflops(
            scaled_spatial_size, output_channels, output_channels, 3
        )
    total_cost += convolution_gflops(input_resolution, 64, 1, 1)
    return total_cost


def attention_decoder_gflops(input_resolution: int = 512) -> float:
    """Return dense decoder FLOPs plus the four soft attention gates."""
    total_cost = static_decoder_gflops(input_resolution)
    for spatial_size, _, output_channels in _DECODER_STAGE_SPECS:
        scaled_spatial_size = spatial_size * input_resolution // 512
        intermediate_channels = output_channels // 2
        total_cost += convolution_gflops(
            scaled_spatial_size, output_channels, intermediate_channels, 1
        )
        total_cost += convolution_gflops(
            scaled_spatial_size, output_channels, intermediate_channels, 1
        )
        total_cost += convolution_gflops(
            scaled_spatial_size, intermediate_channels, 1, 1
        )
    return total_cost


def dynamic_decoder_gflops(
    active_fractions: Sequence[float],
    dilated_active_fractions: Sequence[float] | None = None,
    conditional_low_entropy: bool = True,
) -> float:
    """Return nominal or halo-adjusted FLOPs for the Dynamic U-Net decoder."""
    if len(active_fractions) != len(_GATED_STAGE_SPECS):
        raise ValueError("Expected one active fraction for each of three gated stages")
    if dilated_active_fractions is not None and len(dilated_active_fractions) != len(
        _GATED_STAGE_SPECS
    ):
        raise ValueError("Expected one dilated active fraction for each gated stage")
    if any(not 0.0 <= value <= 1.0 for value in active_fractions):
        raise ValueError("Active fractions must be in the range [0, 1]")
    if dilated_active_fractions is not None and any(
        not 0.0 <= value <= 1.0 for value in dilated_active_fractions
    ):
        raise ValueError("Dilated active fractions must be in the range [0, 1]")

    coarse_resolution, coarse_input_channels, coarse_output_channels = _DECODER_STAGE_SPECS[0]
    total_cost = decoder_transpose_gflops(
        coarse_resolution, coarse_input_channels, coarse_output_channels
    )
    total_cost += convolution_gflops(
        coarse_resolution, coarse_output_channels * 2, coarse_output_channels, 3
    )
    total_cost += convolution_gflops(
        coarse_resolution, coarse_output_channels, coarse_output_channels, 3
    )

    for stage_index, ((spatial_size, input_channels, output_channels), active_fraction) in enumerate(
        zip(_GATED_STAGE_SPECS, active_fractions)
    ):
        total_cost += decoder_transpose_gflops(spatial_size, input_channels, output_channels)
        total_cost += convolution_gflops(spatial_size, output_channels, 1, 1)
        total_cost += convolution_gflops(spatial_size, output_channels * 2, output_channels, 1)
        convolution1_fraction = (
            dilated_active_fractions[stage_index]
            if dilated_active_fractions is not None
            else active_fraction
        )
        total_cost += convolution1_fraction * convolution_gflops(
            spatial_size, output_channels, output_channels, 3
        )
        total_cost += active_fraction * convolution_gflops(
            spatial_size, output_channels, output_channels, 3
        )
        projection_fraction = (1.0 - active_fraction) if conditional_low_entropy else 1.0
        total_cost += projection_fraction * convolution_gflops(
            spatial_size, output_channels, output_channels, 1
        )
    total_cost += convolution_gflops(512, 64, 1, 1)
    return total_cost


def count_parameters(model: nn.Module) -> dict[str, int]:
    """Count all model parameters and the subset that can receive gradients."""
    return {
        "total_params": sum(parameter.numel() for parameter in model.parameters()),
        "trainable_params": sum(
            parameter.numel() for parameter in model.parameters() if parameter.requires_grad
        ),
    }


def compute_dilated_routing_mask(routing_mask: Tensor) -> Tensor:
    """Expand active routes by the 3x3 convolution neighborhood halo."""
    return functional.max_pool2d(
        routing_mask.float(), kernel_size=3, stride=1, padding=1
    ) > 0

===== src/losses.py =====
"""Binary cross-entropy, Dice, and dynamic routing objectives."""

from __future__ import annotations

from collections.abc import Sequence

import torch
import torch.nn.functional as functional
from torch import Tensor, nn


class BCEDiceLoss(nn.Module):
    """Average binary cross-entropy and smoothed soft Dice losses."""

    def __init__(self, smooth: float = 1e-6) -> None:
        super().__init__()
        self.binary_cross_entropy = nn.BCEWithLogitsLoss()
        self.smooth = smooth

    def forward(self, logits: Tensor, targets: Tensor) -> Tensor:
        targets = targets.to(dtype=logits.dtype)
        binary_cross_entropy_loss = self.binary_cross_entropy(logits, targets)
        dice_targets = targets.float()
        probabilities = torch.sigmoid(logits.float())
        intersection = torch.sum(probabilities * dice_targets, dim=(1, 2, 3))
        total_mass = torch.sum(probabilities + dice_targets, dim=(1, 2, 3))
        dice_score = (2.0 * intersection + self.smooth) / (total_mass + self.smooth)
        dice_loss = 1.0 - dice_score.mean()
        return 0.5 * binary_cross_entropy_loss + 0.5 * dice_loss


class DynamicUNetLoss(nn.Module):
    """Combine primary, auxiliary deep supervision, and routing sparsity losses."""

    def __init__(
        self,
        auxiliary_weight: float = 0.2,
        flops_weight: float = 0.05,
        smooth: float = 1e-6,
    ) -> None:
        super().__init__()
        self.segmentation_loss = BCEDiceLoss(smooth=smooth)
        self.auxiliary_weight = auxiliary_weight
        self.flops_weight = flops_weight

    def forward(
        self,
        logits: Tensor,
        targets: Tensor,
        auxiliary_logits_list: Sequence[Tensor] | None = None,
        routing_masks_list: Sequence[Tensor] | None = None,
    ) -> Tensor:
        primary_loss = self.segmentation_loss(logits, targets)
        auxiliary_losses = []
        for auxiliary_logits in auxiliary_logits_list or ():
            resized_targets = functional.interpolate(
                targets.float(),
                size=auxiliary_logits.shape[-2:],
                mode="nearest",
            )
            auxiliary_losses.append(self.segmentation_loss(auxiliary_logits, resized_targets))
        if auxiliary_losses:
            auxiliary_loss = torch.stack(auxiliary_losses).mean()
        else:
            auxiliary_loss = primary_loss.new_zeros(())
        if routing_masks_list:
            flops_penalty = torch.stack(
                [routing_mask.to(dtype=logits.dtype).mean() for routing_mask in routing_masks_list]
            ).mean()
        else:
            flops_penalty = primary_loss.new_zeros(())
        return (
            primary_loss
            + self.auxiliary_weight * auxiliary_loss
            + self.flops_weight * flops_penalty
        )

===== src/metrics.py =====
"""Micro segmentation metrics derived from binary confusion counts."""

from __future__ import annotations

import torch
from torch import Tensor


def compute_confusion_counts(
    logits: Tensor, targets: Tensor, threshold: float = 0.5
) -> dict[str, int]:
    """Compute pixel-level TP, FP, FN, and TN counts for a binary mask."""
    predictions = torch.sigmoid(logits) >= threshold
    ground_truth = targets >= 0.5
    return {
        "true_positives": int((predictions & ground_truth).sum().item()),
        "false_positives": int((predictions & ~ground_truth).sum().item()),
        "false_negatives": int((~predictions & ground_truth).sum().item()),
        "true_negatives": int((~predictions & ~ground_truth).sum().item()),
    }


def metrics_from_counts(counts: dict[str, int]) -> dict[str, float]:
    """Compute positive-class IoU, F1, precision, and recall from counts."""
    true_positives = counts["true_positives"]
    false_positives = counts["false_positives"]
    false_negatives = counts["false_negatives"]
    if true_positives == 0 and false_positives == 0 and false_negatives == 0:
        return {"iou": 1.0, "f1_score": 1.0, "precision": 1.0, "recall": 1.0}
    iou_denominator = true_positives + false_positives + false_negatives
    f1_denominator = 2 * true_positives + false_positives + false_negatives
    precision_denominator = true_positives + false_positives
    recall_denominator = true_positives + false_negatives
    return {
        "iou": true_positives / iou_denominator if iou_denominator else 0.0,
        "f1_score": 2 * true_positives / f1_denominator if f1_denominator else 0.0,
        "precision": true_positives / precision_denominator if precision_denominator else 0.0,
        "recall": true_positives / recall_denominator if recall_denominator else 0.0,
    }


def compute_segmentation_metrics(
    logits: Tensor, targets: Tensor, threshold: float = 0.5
) -> dict[str, float]:
    """Compute pixel-level IoU, F1, precision, and recall for a binary mask."""
    return metrics_from_counts(compute_confusion_counts(logits, targets, threshold))

def _seed_everything(seed: int = RANDOM_SEED, deterministic: bool = False) -> None:
    """Seed Python, NumPy, and PyTorch random number generators."""
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)
    if torch.cuda.is_available():
        torch.cuda.manual_seed(seed)
        torch.cuda.manual_seed_all(seed)
    if hasattr(torch.backends, "cudnn"):
        torch.backends.cudnn.deterministic = deterministic
        torch.backends.cudnn.benchmark = not deterministic


def _build_model(model_name: str, threshold: float) -> nn.Module:
    """Construct the requested baseline or dynamic U-Net model."""
    if model_name == "static_unet":
        return StaticUNet()
    if model_name == "attention_unet":
        return SoftAttentionUNet()
    if model_name == "dynamic_unet":
        return DynamicUNet(threshold=threshold)
    raise ValueError(f"Unsupported model: {model_name}")


def _forward_with_loss(
    model: nn.Module,
    criterion: BCEDiceLoss | DynamicUNetLoss,
    inputs: Tensor,
    targets: Tensor,
    is_dynamic_model: bool,
) -> tuple[Tensor, Tensor, list[Tensor], list[Tensor]]:
    """Run a model and its matching segmentation objective."""
    if is_dynamic_model:
        logits, auxiliary_logits_list, routing_masks_list = model(
            inputs,
            return_internals=True,
        )
        loss = criterion(
            logits,
            targets,
            auxiliary_logits_list,
            routing_masks_list,
        )
        return logits, loss, auxiliary_logits_list, routing_masks_list
    logits = model(inputs)
    loss = criterion(logits, targets)
    return logits, loss, [], []

    accumulation_steps: int,
) -> float:
    """Train for one epoch and return the mean sample-weighted loss."""
    if accumulation_steps <= 0:
        raise ValueError("accumulation_steps must be a positive integer")
    model.train()
    total_loss = 0.0
    sample_count = 0
    total_batches = len(data_loader)
    progress_bar = tqdm(
        data_loader,
        desc=f"Train {epoch_index}/{epoch_count}",
        leave=False,
    )
    for batch_index, (inputs, targets) in enumerate(progress_bar):
        inputs = inputs.to(device, non_blocking=True)
        targets = targets.to(device, non_blocking=True)
        with torch.amp.autocast("cuda", enabled=device.type == "cuda"):
            _, loss, _, _ = _forward_with_loss(
                model,
                criterion,
                inputs,
                targets,
                is_dynamic_model,
            )
        accumulated_loss = loss / accumulation_steps
        scaler.scale(accumulated_loss).backward()
        is_accumulation_boundary = (batch_index + 1) % accumulation_steps == 0
        is_last_batch = batch_index + 1 == total_batches
        if is_accumulation_boundary or is_last_batch:
            scaler.step(optimizer)
            scaler.update()
            optimizer.zero_grad(set_to_none=True)
        batch_size = inputs.shape[0]
        total_loss += float(loss.detach().item()) * batch_size
        sample_count += batch_size
        progress_bar.set_postfix(loss=f"{loss.detach().item():.4f}")
    return total_loss / max(sample_count, 1)


@torch.no_grad()
def validate_one_epoch(
    model: nn.Module,
    data_loader: torch.utils.data.DataLoader,
    criterion: BCEDiceLoss | DynamicUNetLoss,
    device: torch.device,
    epoch_index: int,
    epoch_count: int,
    is_dynamic_model: bool,
) -> dict[str, float]:
    """Evaluate loss, segmentation metrics, and dynamic routing utilization."""
    model.eval()
    total_loss = 0.0
    sample_count = 0
    pooled_counts = {
        "true_positives": 0,
        "false_positives": 0,
        "false_negatives": 0,
        "true_negatives": 0,
    }
    macro_f1_total = 0.0
    active_percentage_total = 0.0
    progress_bar = tqdm(
        data_loader,
        desc=f"Valid {epoch_index}/{epoch_count}",
        leave=False,
    )
    for inputs, targets in progress_bar:
        inputs = inputs.to(device, non_blocking=True)
        targets = targets.to(device, non_blocking=True)
        with torch.amp.autocast("cuda", enabled=device.type == "cuda"):
            logits, loss, _, routing_masks_list = _forward_with_loss(
                model,
                criterion,
                inputs,
                targets,
                is_dynamic_model,
            )
        batch_size = inputs.shape[0]
        total_loss += float(loss.item()) * batch_size
        batch_counts = compute_confusion_counts(logits.float(), targets.float())
        for metric_name, metric_value in batch_counts.items():
            pooled_counts[metric_name] += metric_value
        macro_f1_total += metrics_from_counts(batch_counts)["f1_score"] * batch_size
        if routing_masks_list:
            batch_active_percentage = 100.0 * torch.stack(
                [routing_mask.float().mean() for routing_mask in routing_masks_list]
            ).mean()
            active_percentage_total += float(batch_active_percentage.item()) * batch_size
        sample_count += batch_size
        progress_bar.set_postfix(loss=f"{loss.item():.4f}")
    denominator = max(sample_count, 1)
    results = {
        "loss": total_loss / denominator,
        **metrics_from_counts(pooled_counts),
        "macro_f1_score": macro_f1_total / denominator,
        "active_percentage": active_percentage_total / denominator,
    }
    return results


def _append_epoch_history(
    log_path: Path, row: dict[str, object], overwrite: bool = False
) -> None:
    """Append one epoch's metrics to the model-specific CSV history."""
    log_path.parent.mkdir(parents=True, exist_ok=True)
    write_header = overwrite or not log_path.exists()
    mode = "w" if overwrite else "a"
    with log_path.open(mode, newline="", encoding="utf-8") as log_file:
        writer = csv.DictWriter(log_file, fieldnames=list(row))
        if write_header:
            writer.writeheader()
        writer.writerow(row)


def parse_args(arguments: Sequence[str] | None = None) -> argparse.Namespace:
    """Parse the command-line options for a training run."""
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "--model",
        choices=("static_unet", "attention_unet", "dynamic_unet"),
        default="dynamic_unet",
    )
    parser.add_argument("--epochs", type=int, default=20)
    parser.add_argument("--batch_size", "--batch-size", type=int, default=4)
    parser.add_argument("--learning_rate", "--learning-rate", type=float, default=1e-4)
    parser.add_argument("--flops_weight", "--flops-weight", type=float, default=0.05)
    parser.add_argument("--threshold", type=float, default=0.5)
    parser.add_argument("--num_workers", "--num-workers", type=int, default=4)
    parser.add_argument("--accumulation_steps", "--accumulation-steps", type=int, default=4)
    parsed_arguments = parser.parse_args(arguments)
    if parsed_arguments.epochs <= 0:
        parser.error("--epochs must be positive")
    if parsed_arguments.batch_size <= 0:
        parser.error("--batch_size must be positive")
    if parsed_arguments.learning_rate <= 0.0:
        parser.error("--learning_rate must be positive")
    if parsed_arguments.flops_weight < 0.0:
        parser.error("--flops_weight must be nonnegative")
    if parsed_arguments.num_workers < 0:
        parser.error("--num_workers must be nonnegative")
    if parsed_arguments.accumulation_steps <= 0:
        parser.error("--accumulation_steps must be positive")
    if not 0.0 <= parsed_arguments.threshold <= 1.0:
        parser.error("--threshold must be in the range [0, 1]")
    return parsed_arguments


def main(arguments: Sequence[str] | None = None) -> None:
    """Run the configured training and validation epochs."""
    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s | %(levelname)s | %(name)s | %(message)s",
    )
    options = parse_args(arguments)
    _seed_everything()
    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    train_loader, validation_loader, _ = get_dataloaders(
        batch_size=options.batch_size,
        num_workers=options.num_workers,
    )
    model = _build_model(options.model, options.threshold).to(device)
    is_dynamic_model = options.model == "dynamic_unet"
    if is_dynamic_model:
        criterion: BCEDiceLoss | DynamicUNetLoss = DynamicUNetLoss(
            flops_weight=options.flops_weight
        )
    else:
        criterion = BCEDiceLoss()
    optimizer = torch.optim.AdamW(model.parameters(), lr=options.learning_rate)
    scaler = torch.amp.GradScaler("cuda", enabled=device.type == "cuda")
    checkpoint_dir = Path("models")
    checkpoint_dir.mkdir(parents=True, exist_ok=True)
    checkpoint_path = checkpoint_dir / f"best_{options.model}.pth"
    history_path = Path("results") / "tables" / f"{options.model}_training_log.csv"
    best_validation_f1 = float("-inf")

    LOGGER.info("Training %s on %s", options.model, device)
    for epoch_index in range(1, options.epochs + 1):
        training_loss = train_one_epoch(
            model,
            train_loader,
            criterion,
            optimizer,
            scaler,
            device,
            epoch_index,
            options.epochs,
            is_dynamic_model,
            options.accumulation_steps,
        )
        validation_results = validate_one_epoch(
            model,
            validation_loader,
            criterion,
            device,
            epoch_index,
            options.epochs,
            is_dynamic_model,
        )
        LOGGER.info(
            "Epoch %d/%d | train_loss=%.6f | val_loss=%.6f | IoU=%.6f | F1=%.6f | precision=%.6f | recall=%.6f%s",
            epoch_index,
            options.epochs,
            training_loss,
            validation_results["loss"],
            validation_results["iou"],
            validation_results["f1_score"],
            validation_results["precision"],
            validation_results["recall"],
            f" | active_3x3={validation_results['active_percentage']:.2f}%"
            if is_dynamic_model
            else "",
        )
        history_row: dict[str, object] = {
            "epoch": epoch_index,
            "train_loss": training_loss,
            "validation_loss": validation_results["loss"],
            "validation_iou": validation_results["iou"],
            "validation_f1": validation_results["f1_score"],
            "validation_macro_f1": validation_results["macro_f1_score"],
            "validation_precision": validation_results["precision"],
            "validation_recall": validation_results["recall"],
            "active_3x3_percentage": validation_results["active_percentage"]
            if is_dynamic_model
            else "",
        }
        _append_epoch_history(history_path, history_row, overwrite=epoch_index == 1)
        if validation_results["f1_score"] > best_validation_f1:
            best_validation_f1 = validation_results["f1_score"]
            torch.save(
                {
                    "epoch": epoch_index,
                    "model_name": options.model,
                    "model_state_dict": model.state_dict(),
                    "optimizer_state_dict": optimizer.state_dict(),
                    "validation_metrics": validation_results,
                    "random_seed": RANDOM_SEED,
                    "threshold": options.threshold,
                    "flops_weight": options.flops_weight,
                    "cli_args": vars(options),
                },
                checkpoint_path,
            )
            LOGGER.info("Saved best checkpoint to %s", checkpoint_path)


if __name__ == "__main__":
    main()

"""Paired segmentation transforms with training-fold RGB normalization."""

from __future__ import annotations

import random
from dataclasses import dataclass

import numpy as np
import torch
from PIL import Image
from torchvision.transforms import InterpolationMode
from torchvision.transforms.v2 import functional as v2_functional

# Found through eda loop ran before writing this file
TRAIN_MEAN = (0.43380652, 0.43693942, 0.40113835)
TRAIN_STD = (0.29580833, 0.28946298, 0.30155425)


@dataclass(frozen=True)
class SegmentationTransform:
    """Convert and normalize paired PIL data, optionally augmenting it jointly.

    Spatial operations use torchvision.transforms.v2 functional transforms
    with shared random decisions. Quarter-turn rotations use exact tensor
    permutations so neither image nor binary mask receives interpolation.
    """

    training: bool

    def __call__(self, image: Image.Image, mask: Image.Image) -> tuple[torch.Tensor, torch.Tensor]:
        """Transform an RGB image and grayscale mask into normalized tensors."""
        image_array = np.asarray(image.convert("RGB"), dtype=np.uint8).copy()
        mask_array = np.asarray(mask.convert("L"), dtype=np.uint8).copy()
        image_tensor = torch.from_numpy(image_array).permute(2, 0, 1).to(torch.float32).div_(255.0)
        mask_tensor = torch.from_numpy(mask_array).unsqueeze(0).to(torch.float32)
        image_tensor = v2_functional.resize(
            image_tensor,
            size=[512, 512],
            interpolation=InterpolationMode.BILINEAR,
        )
        mask_tensor = v2_functional.resize(
            mask_tensor,
            size=[512, 512],
            interpolation=InterpolationMode.NEAREST,
        )
        mask_tensor = (mask_tensor > 0).to(torch.float32)

        if image_tensor.shape[-2:] != mask_tensor.shape[-2:]:
            raise ValueError(
                "Image and mask dimensions differ: "
                f"{tuple(image_tensor.shape[-2:])} vs {tuple(mask_tensor.shape[-2:])}"
            )

        if self.training:
            if random.random() < 0.5:
                image_tensor = v2_functional.horizontal_flip(image_tensor)
                mask_tensor = v2_functional.horizontal_flip(mask_tensor)
            if random.random() < 0.5:
                image_tensor = v2_functional.vertical_flip(image_tensor)
                mask_tensor = v2_functional.vertical_flip(mask_tensor)
            if random.random() < 0.5:
                quarter_turns = random.choice((1, 2, 3))
                image_tensor = torch.rot90(image_tensor, k=quarter_turns, dims=(-2, -1))
                mask_tensor = torch.rot90(mask_tensor, k=quarter_turns, dims=(-2, -1))

        mean = torch.tensor(TRAIN_MEAN, dtype=image_tensor.dtype).view(3, 1, 1)
        std = torch.tensor(TRAIN_STD, dtype=image_tensor.dtype).view(3, 1, 1)
        image_tensor = (image_tensor - mean) / std
        mask_tensor = (mask_tensor > 0.5).to(torch.float32)
        return image_tensor.to(torch.float32), mask_tensor.to(torch.float32)


def get_train_transforms() -> SegmentationTransform:
    """Return paired random flips/quarter-turn rotation and normalization."""
    return SegmentationTransform(training=True)


def get_val_test_transforms() -> SegmentationTransform:
    """Return deterministic tensor conversion and training-fold normalization."""
    return SegmentationTransform(training=False)

def measure_convolution_gflops(model, sample_tensor):
    measured_flops = {"value": 0.0}
    hook_handles = []

    def count_convolution(module, module_inputs, module_output):
        if isinstance(module_output, torch.Tensor):
            output_height, output_width = module_output.shape[-2:]
            operations = (
                2
                * output_height
                * output_width
                * module.in_channels
                * module.out_channels
                * module.kernel_size[0]
                * module.kernel_size[1]
                / module.groups
            )
            measured_flops["value"] += operations / 1e9

    for module in model.modules():
        if isinstance(module, nn.Conv2d):
            hook_handles.append(module.register_forward_hook(count_convolution))
    with torch.no_grad():
        model(sample_tensor)
    for hook_handle in hook_handles:
        hook_handle.remove()
    return measured_flops["value"]
